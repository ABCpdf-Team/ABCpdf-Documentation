    /// <summary>The height of the resulting image data</summary>
    [MarshalAs(UnmanagedType.U4)]
    public UInt32 height;
  }

  /// <summary>Info about the active camera</summary>
  [StructLayout(LayoutKind.Sequential, CharSet = CharSet.Unicode)]
  public struct ActiveCameraInfo
  {
    /// <summary>The name of the camera</summary>
    public IntPtr cameraName;
    /// <summary>The name length of the camera</summary>
    [MarshalAs(UnmanagedType.I4)]
    public Int32 cameraNameLen;
    /// <summary>The camera mode of operation, see ECameraMode</summary>
    [MarshalAs(UnmanagedType.U4)]
    public UInt32 cameraMode;
    /// <summary>Set if the camera uses orthogonal projection</summary>
    [MarshalAs(UnmanagedType.U4)]
    public UInt32 isOrtho;
    /// <summary>Set if the camera can be modified at all</summary>
    [MarshalAs(UnmanagedType.U4)]
    public UInt32 isModifiable;
  }

  /// <summary>Info about the loading progress, sent in the LoadProgress event</summary>
  [StructLayout(LayoutKind.Sequential, CharSet = CharSet.Unicode)]
  public struct ProgressInfo
  {
    /// <summary>The percentage of the current loading process</summary>
    [MarshalAs(UnmanagedType.R4)]
    public float currentProgress;
    /// <summary>The user data that was passed in with LoadSceneParams</summary>
    public IntPtr userData;
  }
